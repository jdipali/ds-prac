import random


TARGET = "10101010"
GENE_LENGTH = len(TARGET)
POPULATION_SIZE = 100
MUTATION_RATE = 0.05
GENERATIONS = 200
ELITISM_COUNT = 2


def calculate_fitness(individual):
    fitness = 0

    for i in range(GENE_LENGTH):
        if individual[i] == TARGET[i]:
            fitness += 1

    return fitness


def create_individual():
    return "".join(random.choice("01") for _ in range(GENE_LENGTH))


def create_initial_population():
    return [create_individual() for _ in range(POPULATION_SIZE)]


def selection(population, fitnesses):
    tournament_size = 5

    competitors_indices = random.sample(
        range(POPULATION_SIZE),
        tournament_size,
    )

    best_index = -1
    best_fitness = -1

    for index in competitors_indices:
        if fitnesses[index] > best_fitness:
            best_fitness = fitnesses[index]
            best_index = index

    return population[best_index]


def crossover(parent1, parent2):
    crossover_point = random.randint(1, GENE_LENGTH - 1)

    child1 = (
        parent1[:crossover_point]
        + parent2[crossover_point:]
    )

    child2 = (
        parent2[:crossover_point]
        + parent1[crossover_point:]
    )

    return child1, child2


def mutate(individual):
    mutated_list = list(individual)

    for i in range(GENE_LENGTH):
        if random.random() < MUTATION_RATE:
            mutated_list[i] = (
                "1"
                if mutated_list[i] == "0"
                else "0"
            )

    return "".join(mutated_list)


def run_genetic_algorithm():
    population = create_initial_population()

    print(f"Target String: {TARGET}")
    print("---------------------------------")

    for generation in range(1, GENERATIONS + 1):
        evaluated_population = []

        for individual in population:
            fitness = calculate_fitness(individual)
            evaluated_population.append(
                (individual, fitness)
            )

        evaluated_population.sort(
            key=lambda x: x[1],
            reverse=True,
        )

        best_individual, best_fitness = (
            evaluated_population[0]
        )

        if (
            generation % 20 == 0
            or best_fitness == GENE_LENGTH
        ):
            print(
                f"Gen {generation:3d}: "
                f"Best Fit = {best_fitness}/{GENE_LENGTH}, "
                f"Individual = {best_individual}"
            )

        if best_fitness == GENE_LENGTH:
            print("\nTarget found! Stopping GA.")
            break

        new_population = []

        for i in range(ELITISM_COUNT):
            new_population.append(
                evaluated_population[i][0]
            )

        while len(new_population) < POPULATION_SIZE:
            current_population = [
                item[0]
                for item in evaluated_population
            ]

            current_fitnesses = [
                item[1]
                for item in evaluated_population
            ]

            parent1 = selection(
                current_population,
                current_fitnesses,
            )

            parent2 = selection(
                current_population,
                current_fitnesses,
            )

            child1, child2 = crossover(
                parent1,
                parent2,
            )

            child1 = mutate(child1)
            child2 = mutate(child2)

            new_population.append(child1)

            if len(new_population) < POPULATION_SIZE:
                new_population.append(child2)

        population = new_population


if __name__ == "__main__":
    run_genetic_algorithm()

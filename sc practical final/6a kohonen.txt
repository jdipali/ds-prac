import numpy as np
import matplotlib.pyplot as plt


class SOM:
    """
    A simple implementation of a Kohonen Self-Organizing Map (SOM).
    """

    def __init__(
        self,
        input_dim,
        map_width,
        map_height,
        initial_lr=0.5,
        initial_radius=None,
    ):
        self.input_dim = input_dim
        self.map_shape = (map_width, map_height)
        self.initial_lr = initial_lr

        if initial_radius is None:
            self.initial_radius = max(map_width, map_height) / 2.0
        else:
            self.initial_radius = initial_radius

        self.weights = np.random.rand(map_width, map_height, input_dim)

        self.neuron_coords = np.array(
            [[i, j] for i in range(map_width) for j in range(map_height)]
        )

        print(
            f"SOM initialized: {map_width}x{map_height} map, Input Dim: {input_dim}"
        )

    def _find_bmu(self, input_vector):
        distances = np.linalg.norm(self.weights - input_vector, axis=2)

        bmu_index_flat = np.argmin(distances)

        bmu_coords = np.unravel_index(bmu_index_flat, self.map_shape)

        return bmu_coords

    def _update_weights(
        self,
        input_vector,
        bmu_coords,
        iteration,
        max_iterations,
    ):
        time_constant = max_iterations / np.log(self.initial_radius)

        current_radius = self.initial_radius * np.exp(
            -iteration / time_constant
        )
        radius_sq = current_radius**2

        current_lr = self.initial_lr * np.exp(
            -iteration / max_iterations
        )

        bmu_x, bmu_y = bmu_coords
        bmu_coord_vec = np.array([bmu_x, bmu_y])

        dist_sq_from_bmu = np.sum(
            (self.neuron_coords - bmu_coord_vec) ** 2,
            axis=1,
        )

        neighborhood_func = np.exp(
            -dist_sq_from_bmu / (2 * radius_sq)
        )

        neighborhood_map = neighborhood_func.reshape(
            self.map_shape[0],
            self.map_shape[1],
            1,
        )

        weight_diff = input_vector - self.weights

        delta_w = current_lr * neighborhood_map * weight_diff

        self.weights += delta_w

    def train(self, data, num_epochs):
        max_iterations = num_epochs * data.shape[0]
        iteration = 0

        print(
            f"\nStarting training for {num_epochs} epochs "
            f"({max_iterations} updates)..."
        )

        for epoch in range(num_epochs):
            np.random.shuffle(data)

            for input_vector in data:
                bmu_coords = self._find_bmu(input_vector)

                self._update_weights(
                    input_vector,
                    bmu_coords,
                    iteration,
                    max_iterations,
                )

                iteration += 1

        print("Training finished.")


if __name__ == "__main__":
    np.random.seed(42)

    c1 = np.random.rand(50, 2) * 2 + np.array([1, 1])
    c2 = np.random.rand(50, 2) * 2 + np.array([6, 6])
    c3 = np.random.rand(50, 2) * 2 + np.array([3, 8])

    data = np.vstack((c1, c2, c3))

    data_normalized = (
        data - np.min(data, axis=0)
    ) / (
        np.max(data, axis=0)
        - np.min(data, axis=0)
    )

    INPUT_DIM = data_normalized.shape[1]
    MAP_WIDTH = 10
    MAP_HEIGHT = 10
    NUM_EPOCHS = 50

    som_net = SOM(
        input_dim=INPUT_DIM,
        map_width=MAP_WIDTH,
        map_height=MAP_HEIGHT,
        initial_lr=0.8,
        initial_radius=5.0,
    )

    som_net.train(data_normalized, NUM_EPOCHS)

    final_weights = som_net.weights.reshape(-1, INPUT_DIM)

    plt.figure(figsize=(10, 8))

    plt.scatter(
        data_normalized[:, 0],
        data_normalized[:, 1],
        s=50,
        c="gray",
        label="Input Data Points",
        alpha=0.6,
    )

    plt.scatter(
        final_weights[:, 0],
        final_weights[:, 1],
        s=150,
        c="r",
        marker="o",
        label="SOM Neuron Weights",
    )

    for i in range(MAP_WIDTH):
        for j in range(MAP_HEIGHT):
            w = som_net.weights[i, j]

            if i + 1 < MAP_WIDTH:
                w_right = som_net.weights[i + 1, j]
                plt.plot(
                    [w[0], w_right[0]],
                    [w[1], w_right[1]],
                    "r-",
                    linewidth=1,
                )

            if j + 1 < MAP_HEIGHT:
                w_top = som_net.weights[i, j + 1]
                plt.plot(
                    [w[0], w_top[0]],
                    [w[1], w_top[1]],
                    "r-",
                    linewidth=1,
                )

    plt.title("Kohonen Self-Organizing Map (SOM) Clustering")
    plt.xlabel("Feature 1 (Normalized)")
    plt.ylabel("Feature 2 (Normalized)")
    plt.legend()
    plt.grid(True, linestyle=":")
    plt.show()
